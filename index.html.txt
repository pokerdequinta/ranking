    <script>
        let players = [];
        
        // Carregar dados
        async function loadData() {
            try {
                const result = await window.storage.get('poker-players');
                if (result && result.value) {
                    players = JSON.parse(result.value);
                }
            } catch (e) {
                players = [];
            }
            updateAll();
        }
        
        // Salvar dados
        async function saveData() {
            try {
                await window.storage.set('poker-players', JSON.stringify(players));
            } catch (e) {
                console.error('Erro ao salvar:', e);
            }
        }
        
        // Adicionar jogador
        function addPlayer() {
            const name = document.getElementById('playerName').value.trim();
            const photo = document.getElementById('playerPhoto').value.trim();
            
            if (!name) {
                alert('Digite o nome do jogador!');
                return;
            }
            
            const newPlayer = {
                id: Date.now(),
                name: name,
                photo: photo || `https://ui-avatars.com/api/?name=${encodeURIComponent(name)}&size=50&background=d4a121&color=000&bold=true`,
                etapas: Array(12).fill(0),
                totalPoints: 0
            };
            
            players.push(newPlayer);
            saveData();
            updateAll();
            
            document.getElementById('playerName').value = '';
            document.getElementById('playerPhoto').value = '';
        }
        
        // Adicionar pontos
        function addPoints() {
            const playerId = parseInt(document.getElementById('selectPlayer').value);
            const etapa = parseInt(document.getElementById('etapaNum').value);
            const points = parseInt(document.getElementById('pontos').value);
            
            if (!playerId || !etapa || isNaN(points)) {
                alert('Preencha todos os campos!');
                return;
            }
            
            if (etapa < 1 || etapa > 12) {
                alert('Etapa deve ser entre 1 e 12!');
                return;
            }
            
            const player = players.find(p => p.id === playerId);
            if (player) {
                player.etapas[etapa - 1] = points;
                player.totalPoints = player.etapas.reduce((a, b) => a + b, 0);
                saveData();
                updateAll();
                
                document.getElementById('pontos').value = '';
                alert(`${points} pontos registrados para ${player.name} na Etapa ${etapa}!`);
            }
        }
        
        // Remover jogador
        function removePlayer(id) {
            if (confirm('Tem certeza que deseja remover este jogador?')) {
                players = players.filter(p => p.id !== id);
                saveData();
                updateAll();
            }
        }
        
        // Atualizar tudo
        function updateAll() {
            updateRanking();
            updatePlayerSelect();
            updatePlayerListAdmin();
        }
        
        // Atualizar ranking
        function updateRanking() {
            const rankingList = document.getElementById('ranking-list');
            const emptyState = document.getElementById('empty-ranking');
            
            if (players.length === 0) {
                rankingList.style.display = 'none';
                emptyState.style.display = 'block';
                return;
            }
            
            rankingList.style.display = 'flex';
            emptyState.style.display = 'none';
            
            const sorted = [...players].sort((a, b) => b.totalPoints - a.totalPoints);
            
            rankingList.innerHTML = sorted.map((player, index) => {
                const rank = index + 1;
                let rankClass = '';
                let medal = '';
                
                if (rank === 1) {
                    rankClass = 'top3';
                    medal = '🏆';
                } else if (rank === 2) {
                    rankClass = 'top3';
                    medal = '🥈';
                } else if (rank === 3) {
                    rankClass = 'top3';
                    medal = '🥉';
                } else if (rank <= 10) {
                    rankClass = 'top10';
                }
                
                const etapasJogadas = player.etapas.filter(p => p > 0).length;
                
                return `
                    <div class="player-card ${rankClass}">
                        <div class="rank-badge">${rank}º</div>
                        <img src="${player.photo}" alt="${player.name}" class="player-photo">
                        <div class="player-info">
                            <div class="player-name">${player.name}</div>
                            <div class="player-stats">${etapasJogadas} etapas</div>
                        </div>
                        <div class="player-points">${player.totalPoints}${medal ? `<span class="medal">${medal}</span>` : ''}</div>
                    </div>
                `;
            }).join('');
        }
        
        // Atualizar select de jogadores
        function updatePlayerSelect() {
            const select = document.getElementById('selectPlayer');
            select.innerHTML = '<option value="">-- Selecione --</option>' + 
                players.map(p => `<option value="${p.id}">${p.name}</option>`).join('');
        }
        
        // Atualizar lista admin
        function updatePlayerListAdmin() {
            const list = document.getElementById('player-list-admin');
            
            if (players.length === 0) {
                list.innerHTML = '<p style="color: rgba(255,255,255,0.6); text-align: center; padding: 20px;">Nenhum jogador cadastrado</p>';
                return;
            }
            
            list.innerHTML = players.map(p => `
                <div class="player-item-admin">
                    <span>${p.name} (${p.totalPoints} pts)</span>
                    <button class="btn-secondary" onclick="removePlayer(${p.id})">❌</button>
                </div>
            `).join('');
        }
        
        // Alternar abas
        function switchTab(tab) {
            document.querySelectorAll('.tab-btn').forEach(btn => btn.classList.remove('active'));
            document.querySelectorAll('.tab-content').forEach(content => content.classList.remove('active'));
            
            if (tab === 'ranking') {
                document.querySelector('.tab-btn:first-child').classList.add('active');
                document.getElementById('ranking-tab').classList.add('active');
            } else {
                document.querySelector('.tab-btn:last-child').classList.add('active');
                document.getElementById('admin-tab').classList.add('active');
            }
        }
        
        // Inicializar
        loadData();
    </script>
